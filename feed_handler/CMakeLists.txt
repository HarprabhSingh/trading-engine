cmake_minimum_required(VERSION 3.20)
project(feed_handler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_LIVE_FEED "Build experimental Boost/OpenSSL transport" OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
include(CTest)
function(core_target name source)
    add_executable(${name} ${source})
    target_include_directories(${name} PRIVATE include)
    target_link_libraries(${name} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra)
        if(ENABLE_SANITIZERS)
            target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${name} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
core_target(test_decoder tests/test_decoder.cpp)
core_target(test_core tests/test_core.cpp)
core_target(bench_ring tests/bench_ring.cpp)
add_test(NAME decoder COMMAND test_decoder)
add_test(NAME core COMMAND test_core)
set_tests_properties(decoder core PROPERTIES TIMEOUT 60)
if(BUILD_LIVE_FEED)
    find_package(Boost 1.81 REQUIRED COMPONENTS system)
    find_package(OpenSSL REQUIRED)
    core_target(feed_handler src/main.cpp)
    target_link_libraries(feed_handler PRIVATE Boost::system OpenSSL::SSL OpenSSL::Crypto)
endif()
